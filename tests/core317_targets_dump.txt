===== FUNCTION normalizeDungeonRpgRules =====
function normalizeDungeonRpgRules(r){
  const d=defaultDungeonRpgRules(),o={...d,...(r||{})};
  const pos=(k,min=1,max=9999)=>o[k]=Math.max(min,Math.min(max,Number(o[k])||d[k]));
  const non=(k,max=9999)=>o[k]=Math.max(0,Math.min(max,Number(o[k])||0));
  pos("xpPerLevel");non("skillPointsPerLevel",99);non("startingSkillPoints",99);
  pos("physicalDamageStep");non("physicalDamageGain",99);
  o.physicalDamageFormula=o.physicalDamageFormula==="percent"?"percent":"step";non("physicalDamagePercentPerPoint",100);
  pos("magicDamageStep");non("magicDamageGain",99);
  o.magicDamageFormula=o.magicDamageFormula==="percent"?"percent":"step";non("magicDamagePercentPerPoint",100);
  o.rangedDamageFormula=o.rangedDamageFormula==="percent"?"percent":"none";non("rangedDamagePercentPerPoint",100);
  o.hpFormula=o.hpFormula==="percent"?"percent":"step";non("hpPercentPerPoint",100);
  o.manaFormula=o.manaFormula==="percent"?"percent":"step";non("manaPercentPerPoint",100);
  o.critFormula=o.critFormula==="perPoint"?"perPoint":"step";non("critPercentPerPoint",100);
  o.dodgeFormula=o.dodgeFormula==="perPoint"?"perPoint":"step";non("dodgePercentPerPoint",100);
  o.initiativeFormula=o.initiativeFormula==="perPoint"?"perPoint":"step";non("initiativePerPoint",100);
  o.magicResistFormula=o.magicResistFormula==="perPoint"?"perPoint":"step";non("magicResistPerPoint",100);
  o.rulesComplexity=["simple","tactical","rpg","custom"].includes(o.rulesComplexity)?o.rulesComplexity:"simple";
  o.initiativeUsage=["off","turns","tests","both","custom"].includes(o.initiativeUsage)?o.initiativeUsage:"tests";
  o.initiativeTurnMode=["fixed","die","low"].includes(o.initiativeTurnMode)?o.initiativeTurnMode:"fixed";
  o.initiativeDie=[4,6,8,10,12,20,100].includes(Number(o.initiativeDie))?Number(o.initiativeDie):20;
  o.testsEnabled=o.testsEnabled!==false; o.testDie=[4,6,8,10,12,20,100].includes(Number(o.testDie))?Number(o.testDie):20;
  o.testDefaultDifficulty=Math.max(1,Math.min(999,Number(o.testDefaultDifficulty)||10));
  pos("meleeHitStep");non("meleeHitGain",99);
  pos("rangedHitStep");non("rangedHitGain",99);
  pos("magicHitStep");non("magicHitGain",99);
  o.basicAttackBaseChance=Math.max(1,Math.min(99,Number(o.basicAttackBaseChance)||d.basicAttackBaseChance));
  non("baseMana");pos("spiritManaStep");non("manaGain");non("magicAttackManaCost");
  non("baseCrit",100);pos("agilityCritStep");non("critGain",100);non("critCap",100);
  o.critMultiplier=Math.max(1,Math.min(10,Number(o.critMultiplier)||d.critMultiplier));
  non("baseDodge",100);pos("agilityDodgeStep");non("dodgeGain",100);non("dodgeCap",100);
  o.baseInitiative=Math.max(-99,Math.min(999,Number(o.baseInitiative)||0));
  pos("agilityInitiativeStep");non("initiativeGain",99);
  pos("enduranceHpStep");non("hpGain",999);
  non("defensePenaltyPerPoint",25);
  o.hitChanceMin=Math.max(0,Math.min(99,Number.isFinite(Number(o.hitChanceMin))?Number(o.hitChanceMin):d.hitChanceMin));
  o.hitChanceMax=Math.max(o.hitChanceMin,Math.min(100,Number(o.hitChanceMax)||d.hitChanceMax));
  pos("armorReductionStep");non("armorReductionGain",99);non("minPhysicalDamage",99);
  pos("spiritMagicResistStep");non("magicResistGain",99);non("minMagicDamage",99);
  return o;
}

===== FUNCTION dc043ApplyEnemyAttack =====
function dc043ApplyEnemyAttack(inst,def,skill,targetHeroId){
  const profile=dungeonEnemyAttackProfile(def,skill);
  const rules=loadDungeonRpgRules();
  const target=dungeonDerivedForHero(targetHeroId);
  const dice=Math.max(1,Number(profile?.dice)||1);
  let total=0,hits=0,dodges=0,crits=0;
  const rolls=[];

  for(let i=0;i<dice;i++){
    const defPenalty=Math.max(0,
      ((Number(target.defense)||0)-(Number(profile.attributeValue)||0))*
      Number(rules.defensePenaltyPerPoint||0)
    );
    const chance=dungeonClampHitChance((Number(profile.preDefenseChance)||0)-defPenalty);
    const threshold=d100ThresholdFromChance(chance);
    const roll=Math.floor(Math.random()*100)+1;
    const rd={roll,threshold,chance,hit:false,crit:false,dodged:false,damage:0};

    if(roll<threshold){rolls.push(rd);continue}
    hits++; rd.hit=true;

    const critThreshold=Math.max(1,101-(Number(profile.crit)||0));
    const crit=(Number(profile.crit)||0)>0 && roll>=critThreshold;
    if(crit){crits++;rd.crit=true}

    const raw=Math.round((Number(profile.power)||0)*
      (crit?Number(rules.critMultiplier||2):1));
    const mult=gensElementMultiplier(profile.element,gensHeroElementTypes(targetHeroId));
    const elemental=Math.max(0,Math.round(raw*mult));

    const dodge=Math.max(0,Math.min(100,Number(target.dodge)||0));
    if(dodge>0 && (Math.floor(Math.random()*100)+1)>=Math.max(1,101-dodge)){
      dodges++;rd.dodged=true;rolls.push(rd);continue;
    }

    const mit=dungeonMitigationForHero(targetHeroId,profile.damageType);
    const reduction=Math.max(0,Number(mit.reduction)||0);
    const configuredMin=Math.max(0,Number(mit.min)||0);
    const final=elemental>0?Math.max(1,configuredMin,elemental-reduction):0;

    rd.rawDamage=raw;
    rd.elementalDamage=elemental;
    rd.reduction=reduction;
    rd.damage=Math.max(0,final);
    total+=rd.damage;
    rolls.push(rd);
  }

  let applied=0;
  if(total>0){
    try{
      const s=loadState(targetHeroId);
      const maxHp=dungeonMaxHpForHero(targetHeroId,s);
      const beforeWounds=Math.max(0,Number(s.wounds)||0);
      const beforeHp=Math.max(0,maxHp-beforeWounds);
      s.wounds=Math.min(maxHp,beforeWounds+Math.max(0,Math.round(total)));
      localStorage.setItem(key(targetHeroId),JSON.stringify(s));
      applied=Math.max(0,beforeHp-Math.max(0,maxHp-(Number(s.wounds)||0)));
      try{if(typeof z40kSchedulePush==="function")z40kSchedulePush()}catch(e){}
    }catch(e){console.warn("MJ damage 0.43",e)}
  }

  return {
    dice,hits,dodges,crits,total:applied,calculatedTotal:total,
    skill:skill?.name||"Attaque",
    rolls,
    profile:{
      chance:Number(profile?.chance ?? profile?.preDefenseChance ?? 0),
      power:Number(profile?.power||0),
      crit:Number(profile?.crit||0),
      range:profile?.range ?? "—",
      damageType:profile?.damageType||"physique",
      element:profile?.element||"none"
    }
  };
}

  });

  try{return Number(dungeonCombatHeroSnapshot?.(id)?.hp)>0}catch(e){return false}
}
function enemyAlive(id){
  try{
    const e=(loadActiveEnemies?.()||[]).find(x=>String(x.id)===String(id));
    return !!e&&!e.defeated&&Number(e.hp)>0
  }catch(e){return false}
}
function build(){
  const a=[];
  (dungeonCombatSelection?.heroes||[]).forEach(id=>{
    if(!heroAlive(id))return;
    const sn=dungeonCombatHeroSnapshot?.(id)||{};
    a.push({kind:"hero",id:String(id),name:CHARS?.[id]?.name||String(id),initiative:Number(sn.initiative)||0})
  });
  try{
    (loadActiveEnemies?.()||[])
      .filter(e=>(dungeonCombatSelection?.enemies||[]).map(String).includes(String(e.id))&&!e.defeated&&Number(e.hp)>0)
      .forEach(e=>{
        const d=activeEnemyDefinition?.(e.enemyId);
        const dr=dungeonEnemyDerivedForInstance?.(e)||dungeonEnemyDerived?.(d)||{};
        a.push({kind:"enemy",id:String(e.id),name:d?.name||e.name||"Ennemi",initiative:Number(dr.initiative)||0})
      });
  }catch(e){}
  a.sort((x,y)=>y.initiative-x.initiative||(x.kind==="hero"?-1:1));
  return a;
}
function init(force=false){
  const s=sig();
  if(!force&&T.sig===s&&T.order.length)return;
  T={sig:s,order:build(),index:0,aiBusy:false,pendingHero:false};
}
function current(){
  init();
  const alive=build(), keys=new Set(alive.map(a=>a.kind+":"+a.id));
  const old=T.order[T.index];
  T.order=T.order.filter(a=>keys.has(a.kind+":"+a.id));
  alive.forEach(a=>{if(!T.order.some(b=>b.kind===a.kind&&b.id===a.id))T.order.push(a)});
  if(!T.order.length)return null;
  if(old){
    const i=T.order.findIndex(a=>a.kind===old.kind&&a.id===old.id);
    if(i>=0)T.index=i;
  }
  if(T.index>=T.order.length)T.index=0;
  return T.order[T.index]||null;
}
